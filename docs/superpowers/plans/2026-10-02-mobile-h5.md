```ts
import { useSyncExternalStore } from 'react'

/** 订阅 CSS media query;SSR 无关(纯 Vite SPA),getServerSnapshot 仅占位。 */
export function useMediaQuery(query: string): boolean {
  return useSyncExternalStore(
    (cb) => {
      const mql = window.matchMedia(query)
      mql.addEventListener('change', cb)
      return () => mql.removeEventListener('change', cb)
    },
    () => window.matchMedia(query).matches,
    () => false,
  )
}
```

```ts
// src/lib/downscale.ts
// 移动端内存保护:超大图降采样加载。阈值 4096 覆盖 8K 截图/相机原图场景。
export const MAX_IMAGE_EDGE = 4096

export function needsDownscale(w: number, h: number, max = MAX_IMAGE_EDGE): boolean {
  return Math.max(w, h) > max
}

export function scaledSize(w: number, h: number, max = MAX_IMAGE_EDGE): { w: number; h: number } {
  if (!needsDownscale(w, h, max)) return { w, h }
  const scale = max / Math.max(w, h)
  return { w: Math.round(w * scale), h: Math.round(h * scale) }
}

/** canvas 等比缩放并重新解码。PNG 编码保 alpha(JPEG 会丢透明)。
 *  返回的 image 已 onload,可直接上纹理。 */
export function downscaleImage(img: HTMLImageElement, max = MAX_IMAGE_EDGE): Promise<HTMLImageElement> {
  const { w, h } = scaledSize(img.width, img.height, max)
  const c = document.createElement('canvas')
  c.width = w
  c.height = h
  c.getContext('2d')!.drawImage(img, 0, 0, w, h)
  return new Promise((resolve, reject) => {
    const out = new Image()
    out.onload = () => resolve(out)
    out.onerror = () => reject(new Error('downscale decode failed'))
    out.src = c.toDataURL('image/png')
  })
}
```

- [ ] **Step 3: 实现 pinchInfo + hook 重写为 Pointer Events**

在 `useViewport.ts` 追加导出:

```ts
/** 两指从 prev 到 next 的增量:scale = 距离比,anchor = 当前帧中点。
 *  上帧距离为 0 时 scale=1 防除零。 */
export function pinchInfo(prev: [Point, Point], next: [Point, Point]): { scale: number; anchor: Point } {
  const d0 = Math.hypot(prev[1].x - prev[0].x, prev[1].y - prev[0].y)
  const d1 = Math.hypot(next[1].x - next[0].x, next[1].y - next[0].y)
  return {
    scale: d0 === 0 ? 1 : d1 / d0,
    anchor: { x: (next[0].x + next[1].x) / 2, y: (next[0].y + next[1].y) / 2 },
  }
}
```

hook 内事件模型重写(替换现有 `--- 拖动平移 ---` 整段与 `panHandlers`):

```ts
// --- 统一指针:单指/鼠标=平移,双指=pinch 缩放+平移,wheel 保留 ---
// 指针状态:pointerId → 上帧位置。el 内坐标(相对 getBoundingClientRect)。
const pointers = useRef(new Map<number, Point>())
const startPan = useRef<Point | null>(null)
const dragStart = useRef<Point | null>(null)

const onPointerDown = useCallback((e: React.PointerEvent) => {
  if (e.button !== 0 && e.pointerType === 'mouse') return
  const rect = e.currentTarget.getBoundingClientRect()
  pointers.current.set(e.pointerId, { x: e.clientX - rect.left, y: e.clientY - rect.top })
  if (pointers.current.size === 1) {
    dragStart.current = { x: e.clientX, y: e.clientY }
    startPan.current = { x: viewportRef.current.panX, y: viewportRef.current.panY }
    setIsPanning(true)
  } else {
    // 进入双指:清空单指平移基准,避免抬指后跳变
    dragStart.current = null
    startPan.current = null
    setIsPanning(false)
  }
  ;(e.target as Element).setPointerCapture?.(e.pointerId)
}, [])

useEffect(() => {
  const handleMove = (e: PointerEvent) => {
    const rect = elRef.current?.getBoundingClientRect()
    if (!rect || !pointers.current.has(e.pointerId)) return
    const cur: Point = { x: e.clientX - rect.left, y: e.clientY - rect.top }
    const prev = pointers.current.get(e.pointerId)!
    pointers.current.set(e.pointerId, cur)

    if (pointers.current.size === 1 && dragStart.current && startPan.current) {
      // 单指/鼠标平移(屏幕坐标差,与 rect 无关)
      const dx = e.clientX - dragStart.current.x
      const dy = e.clientY - dragStart.current.y
      setViewport({
        ...viewportRef.current,
        panX: startPan.current.x + dx,
        panY: startPan.current.y + dy,
      })
    } else if (pointers.current.size >= 2) {
      const ids = [...pointers.current.keys()].slice(0, 2)
      const pair = ids.map((id) => pointers.current.get(id)!) as [Point, Point]
      const prevPair = [
        ids[0] === e.pointerId ? prev : pair[0],
        ids[1] === e.pointerId ? prev : pair[1],
      ] as [Point, Point]
      const { scale, anchor } = pinchInfo(prevPair, pair)
      setViewport(zoomAtCursor(viewportRef.current, anchor, viewportRef.current.zoom * scale, min, max))
    }
  }
  const handleUp = (e: PointerEvent) => {
    pointers.current.delete(e.pointerId)
    if (pointers.current.size === 0) {
      dragStart.current = null
      startPan.current = null
      setIsPanning(false)
    }
  }
  window.addEventListener('pointermove', handleMove)
  window.addEventListener('pointerup', handleUp)
  window.addEventListener('pointercancel', handleUp)
  return () => {
    window.removeEventListener('pointermove', handleMove)
    window.removeEventListener('pointerup', handleUp)
    window.removeEventListener('pointercancel', handleUp)
  }
}, [min, max])
```

返回值改为 `pointerHandlers: { onPointerDown }`(删除旧 `panHandlers`);`UseViewportResult` 接口同步改名。

```ts
// src/lib/sheetSnap.ts
/** 把手松手时的吸附决策。velocity<0=向上抛。阈值:|v|>600 取速度方向,否则取最近档;
 *  下移超过 closeThreshold 或下移速度>800 判 close。 */
export function settleSnap(
  offsetPx: number,
  velocityPxPerSec: number,
  peekPx: number,
  fullPx: number,
  closeThresholdPx: number,
): 'peek' | 'full' | 'close' {
  if (velocityPxPerSec > 800 || offsetPx > closeThresholdPx) return 'close'
  if (velocityPxPerSec < -600) return 'full'
  const dPeek = Math.abs(offsetPx - peekPx)
  const dFull = Math.abs(offsetPx - fullPx)
  return dPeek <= dFull ? 'peek' : 'full'
}
```

```tsx
// src/components/BottomSheet.tsx
import { useCallback, useEffect, useRef, useState, type ReactNode } from 'react'

/** 移动端底部抽屉:peek(60%)/full(90%) 两档 + 下拉关闭。
 *  把手拖动实时跟手(pointer),松手按 settleSnap 吸附。 */
export default function BottomSheet({ open, onClose, title, full = false, children }: {
  open: boolean
  onClose: () => void
  title?: string
  full?: boolean
  children: ReactNode
}) {
  const [snap, setSnap] = useState<'peek' | 'full'>(full ? 'full' : 'peek')
  const [dragPx, setDragPx] = useState<number | null>(null)
  const last = useRef<{ y: number; t: number } | null>(null)
  const vel = useRef(0)
  const vh = () => window.innerHeight

  const onHandleDown = useCallback((e: React.PointerEvent) => {
    (e.target as Element).setPointerCapture?.(e.pointerId)
    last.current = { y: e.clientY, t: performance.now() }
    vel.current = 0
  }, [])

  useEffect(() => {
    const move = (e: PointerEvent) => {
      if (!last.current) return
      const now = performance.now()
      const dt = now - last.current.t
      if (dt > 0) vel.current = ((e.clientY - last.current.y) / dt) * 1000
      last.current = { y: e.clientY, t: now }
      setDragPx((d) => Math.max(0, (d ?? snapPx(snap, vh())) + e.movementY))
    }
    const up = () => {
      if (!last.current) return
      last.current = null
      const base = snapPx(snap, vh())
      const cur = dragPx ?? base
      const target = settleSnap(cur, vel.current, base * 0.6, base, base * 0.25)
      setDragPx(null)
      if (target === 'close') onClose()
      else setSnap(target)
    }
    window.addEventListener('pointermove', move)
    window.addEventListener('pointerup', up)
    window.addEventListener('pointercancel', up)
    return () => {
      window.removeEventListener('pointermove', move)
      window.removeEventListener('pointerup', up)
      window.removeEventListener('pointercancel', up)
    }
  }, [snap, dragPx, onClose])

  if (!open) return null
  const base = snapPx(snap, vh())
  const height = dragPx !== null ? Math.max(120, base - (dragPx - base * 0)) : base
  // 注:dragPx 直接作为"当前顶边距视口底部偏移"实现简化——执行时以「height = base - movementY 累计」为准,保持跟手。
  return (
    <div className="m-sheet-mask" onClick={onClose}>
      <div className="m-sheet" style={{ height }} onClick={(e) => e.stopPropagation()}>
        <div className="m-sheet-handle" onPointerDown={onHandleDown} />
        {title && <div className="m-sheet-title">{title}</div>}
        <div className="m-sheet-body">{children}</div>
      </div>
    </div>
  )
}

function snapPx(snap: 'peek' | 'full', vh: number): number {
  return Math.round(vh * (snap === 'full' ? 0.9 : 0.6))
}
```

```ts
// src/lib/downloadFlow.ts
import { isWeChatUA } from './clientEnv'

/** 下载路径决策:微信内或移动端 svg → 长按保存灯箱;其余走 a[download]。 */
export function planDownload(env: { wechat?: boolean; isSvg: boolean; forceLightbox?: boolean }): 'anchor' | 'lightbox' {
  const wechat = env.wechat ?? isWeChatUA()
  return wechat || env.forceLightbox ? 'lightbox' : 'anchor'
}

/** 触发浏览器下载(非微信路径)。 */
export function anchorDownload(blob: Blob, fileName: string): void {
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = fileName
  a.click()
  URL.revokeObjectURL(url)
}
```

```ts
// src/lib/svgToPng.ts
/** SVG 字符串 → PNG Blob(canvas 中转)。svg 需自带 width/height 属性。
 *  失败(解码/渲染异常)resolve null,调用方退回原 svg 下载。 */
export function svgToPngBlob(svg: string): Promise<Blob | null> {
  return new Promise((resolve) => {
    const blob = new Blob([svg], { type: 'image/svg+xml;charset=utf-8' })
    const url = URL.createObjectURL(blob)
    const img = new Image()
    img.onload = () => {
      const c = document.createElement('canvas')
      c.width = img.width || 1200
      c.height = img.height || 1600
      const ctx = c.getContext('2d')!
      ctx.fillStyle = '#fff'
      ctx.fillRect(0, 0, c.width, c.height) // PNG 透明背景在相册显示为黑,铺白底
      ctx.drawImage(img, 0, 0)
      URL.revokeObjectURL(url)
      c.toBlob((b) => resolve(b), 'image/png')
    }
    img.onerror = () => { URL.revokeObjectURL(url); resolve(null) }
    img.src = url
  })
}
```
